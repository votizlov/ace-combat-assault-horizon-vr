cmake_minimum_required(VERSION 3.24)
project(acah_vr_poc LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use Visual Studio 2022 on Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(POC_COMPONENT "hook" CACHE STRING "hook (Win32) or host (x64)")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
if(POC_COMPONENT STREQUAL "hook")
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Configure the hook with -A Win32.")
  endif()
  add_library(acah_hook SHARED src/hook/proxy.cpp src/hook/stereo.cpp src/hook/observe.cpp src/hook/hud.cpp src/hook/tree.cpp src/hook/visibility.cpp src/hook/d3d9.def)
  set_target_properties(acah_hook PROPERTIES OUTPUT_NAME d3d9)
  target_include_directories(acah_hook PRIVATE src/common src/hook)
  target_link_libraries(acah_hook PRIVATE bcrypt user32)
elseif(POC_COMPONENT STREQUAL "host")
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Configure the host with -A x64.")
  endif()
  include(FetchContent)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
  set(BUILD_CONFORMANCE_TESTS OFF CACHE BOOL "" FORCE)
  set(BUILD_LOADER ON CACHE BOOL "" FORCE)
  set(DYNAMIC_LOADER ON CACHE BOOL "" FORCE)
  FetchContent_Declare(openxr
    GIT_REPOSITORY https://github.com/KhronosGroup/OpenXR-SDK.git
    GIT_TAG release-1.1.63
    GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(openxr)
  add_executable(acah_vr_host src/host/main.cpp)
  target_include_directories(acah_vr_host PRIVATE src/common)
  target_compile_definitions(acah_vr_host PRIVATE XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11)
  target_link_libraries(acah_vr_host PRIVATE openxr_loader d3d11 dxgi user32)
  add_custom_command(TARGET acah_vr_host POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:openxr_loader> $<TARGET_FILE_DIR:acah_vr_host>)
else()
  message(FATAL_ERROR "POC_COMPONENT must be hook or host.")
endif()
